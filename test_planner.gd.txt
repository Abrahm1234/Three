extends Node

# Use global class_name exports directly (avoid shadowing preloads)

func _ready() -> void:
	var rng := RandomCtx.new()
	add_child(rng)
	rng.reseed(1)

	var bn := ProgramBN.new()
	if bn.has_method("configure_rng"):
		bn.configure_rng(rng)
	var prog := bn.sample({"footprint": Vector2i(18, 14), "bedrooms": 3, "bathrooms": 2, "floors": 1})

	var part := InitBSP.build(prog, 0.1)
	var optimizer := PlanOptimize.new()
	optimizer.iters = 4000
	optimizer.temp0 = 1.0

	var plan_cost := PlanCost.new()
	var allowed_pairs: Array[Vector2i] = []
	var id_to_idx := {}
	for i in range(prog.rooms.size()):
		var prog_id := String(prog.rooms[i].get("id", str(i)))
		id_to_idx[prog_id] = i
	for edge in prog.edges:
		var typ := String(edge.get("type", ""))
		if typ != "door" and typ != "open":
			continue
		var a_idx := int(id_to_idx.get(String(edge.get("a_id", "")), -1))
		var b_idx := int(id_to_idx.get(String(edge.get("b_id", "")), -1))
		if a_idx < 0 or b_idx < 0:
			continue
		allowed_pairs.append(Vector2i(min(a_idx, b_idx), max(a_idx, b_idx)))
	plan_cost.allowed_pairs = allowed_pairs
	optimizer.allowed_pairs = allowed_pairs

	var program_terms := {}
	for room in prog.rooms:
		var label := String(room.get("type", ""))
		if label == "":
			continue
		program_terms[label] = {
			"access_w": float(room.get("access_w", 1.0)),
			"min_exterior": float(room.get("min_exterior", 0.0)),
		}

	var entry_idx := 0
	for i in range(part.rooms.size()):
		if part.rooms[i].label.to_lower() == "entry":
			entry_idx = i
			break

	var result := optimizer.optimize(part, entry_idx, program_terms)
	var access_cost := plan_cost.C_access(result, entry_idx, program_terms)
	var exposure_cost := plan_cost.C_exposure(result, program_terms)
	var total_cost := plan_cost.total(result, entry_idx, program_terms)

	assert(access_cost < 50.0)
	assert(exposure_cost < 200.0)
	assert(total_cost < 1_000_000.0)
	print("OK: ", {"access": access_cost, "exposure": exposure_cost, "total": total_cost})
	get_tree().quit()
